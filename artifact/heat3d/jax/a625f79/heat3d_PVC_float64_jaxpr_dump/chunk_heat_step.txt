let remainder = { lambda ; a:i64[32] b:i64[]. let
    c:i64[] = convert_element_type[new_dtype=int64 weak_type=False] b
    d:bool[] = eq c 0
    e:i64[] = pjit[
      name=_where
      jaxpr={ lambda ; f:bool[] g:i64[] h:i64[]. let
          i:i64[] = select_n f h g
        in (i,) }
    ] d 1 c
    j:i64[32] = rem a e
    k:bool[32] = ne j 0
    l:bool[32] = lt j 0
    m:bool[] = lt e 0
    n:bool[32] = ne l m
    o:bool[32] = and n k
    p:i64[32] = add j e
    q:i64[32] = select_n o j p
  in (q,) } in
