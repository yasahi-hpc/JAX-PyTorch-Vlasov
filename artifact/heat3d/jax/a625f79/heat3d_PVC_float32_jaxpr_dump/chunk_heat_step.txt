let remainder = { lambda ; a:i32[32] b:i32[]. let
    c:i32[] = convert_element_type[new_dtype=int32 weak_type=False] b
    d:bool[] = eq c 0
    e:i32[] = pjit[
      name=_where
      jaxpr={ lambda ; f:bool[] g:i32[] h:i32[]. let
          i:i32[] = select_n f h g
        in (i,) }
    ] d 1 c
    j:i32[32] = rem a e
    k:bool[32] = ne j 0
    l:bool[32] = lt j 0
    m:bool[] = lt e 0
    n:bool[32] = ne l m
    o:bool[32] = and n k
    p:i32[32] = add j e
    q:i32[32] = select_n o j p
  in (q,) } in

    ] t u s
    ba:i32[32,1] = broadcast_in_dim[
      broadcast_dimensions=(0,)
      shape=(32, 1)
      sharding=None
    ] v
